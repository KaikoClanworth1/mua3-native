# Statically recompiled guest code, generated by the recomp tool into MUA3_RECOMP_DIR (never committed: it is
# derived from the game binary). Without it the empty stub table keeps everything on the JIT.
set(MUA3_RECOMP_DIR "${CMAKE_BINARY_DIR}/recomp" CACHE PATH "Directory with generated recomp_*.cpp")
if(EXISTS "${MUA3_RECOMP_DIR}/nxrecomp_full.lib")
    # Full-program output (~1 GB of C++) is compiled outside MSBuild into a prebuilt library: MSBuild loses all
    # progress whenever one of its compilers dies, which happens on shards this large.
    add_library(nxrecomp STATIC IMPORTED GLOBAL)
    set_target_properties(nxrecomp PROPERTIES IMPORTED_LOCATION "${MUA3_RECOMP_DIR}/nxrecomp_full.lib")
    message(STATUS "Recompiled code: prebuilt ${MUA3_RECOMP_DIR}/nxrecomp_full.lib")
else()
    if(EXISTS "${MUA3_RECOMP_DIR}/recomp_table.cpp")
        file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS "${MUA3_RECOMP_DIR}/recomp_*.cpp")
        message(STATUS "Recompiled code: ${MUA3_RECOMP_DIR}")
    else()
        set(RECOMP_SOURCES core/cpu/recomp/recomp_stub.cpp)
    endif()
    add_library(nxrecomp STATIC ${RECOMP_SOURCES})
    target_include_directories(nxrecomp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} "${MUA3_RECOMP_DIR}")
    target_compile_options(nxrecomp PRIVATE /bigobj /GS- /MP6 /Ob3)
    # generated code is huge: debug info would overflow the PDB
    set_target_properties(nxrecomp PROPERTIES MSVC_DEBUG_INFORMATION_FORMAT "$<$<BOOL:0>:Embedded>")
endif()
target_link_libraries(nxcore PUBLIC nxrecomp)
