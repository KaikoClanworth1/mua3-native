# Development only: the dynarmic JIT as a fallback for guest code the recompiler has not covered (MUA3_JIT_FALLBACK=1

# Statically recompiled guest code, generated by the recomp tool into MUA3_RECOMP_DIR (never committed: it is
# derived from the game binary). Without it the empty stub table keeps everything on the JIT.
set(MUA3_RECOMP_DIR "${CMAKE_BINARY_DIR}/recomp" CACHE PATH "Directory with generated recomp_*.cpp")
if(EXISTS "${MUA3_RECOMP_DIR}/nxrecomp_full.lib")
    # Full-program output (~1 GB of C++) is compiled outside MSBuild into a prebuilt library: MSBuild loses all
    # progress whenever one of its compilers dies, which happens on shards this large.
    add_library(nxrecomp STATIC IMPORTED GLOBAL)
    set_target_properties(nxrecomp PROPERTIES IMPORTED_LOCATION "${MUA3_RECOMP_DIR}/nxrecomp_full.lib")
    message(STATUS "Recompiled code: prebuilt ${MUA3_RECOMP_DIR}/nxrecomp_full.lib")
else()
    if(EXISTS "${MUA3_RECOMP_DIR}/recomp_table.cpp")
        file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS "${MUA3_RECOMP_DIR}/recomp_*.cpp")
        message(STATUS "Recompiled code: ${MUA3_RECOMP_DIR}")
    else()
        set(RECOMP_SOURCES core/cpu/recomp/recomp_stub.cpp)
    endif()
    add_library(nxrecomp STATIC ${RECOMP_SOURCES})
    target_include_directories(nxrecomp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} "${MUA3_RECOMP_DIR}")
    target_compile_options(nxrecomp PRIVATE /bigobj /GS- /MP6 /Ob3)
    # generated code is huge: debug info would overflow the PDB
    set_target_properties(nxrecomp PROPERTIES MSVC_DEBUG_INFORMATION_FORMAT "$<$<BOOL:0>:Embedded>")
endif()
target_link_libraries(nxcore PUBLIC nxrecomp)

# AMD FidelityFX FSR 1 (MIT, third_party/ffx_fsr1): its two shader headers embedded as byte arrays, for the HLSL that
# gpu/d3d11/fsr.cpp compiles at runtime.
function(mua3_embed_bytes out_var file)
    file(READ ${file} hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
    string(REGEX REPLACE "((0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],))" "\\1\n" hex "${hex}")
    set(${out_var} "${hex}" PARENT_SCOPE)
endfunction()
set(FFX_FSR1_DIR ${CMAKE_SOURCE_DIR}/third_party/ffx_fsr1)
mua3_embed_bytes(FFX_A_BYTES ${FFX_FSR1_DIR}/ffx_a.h)
mua3_embed_bytes(FFX_FSR1_BYTES ${FFX_FSR1_DIR}/ffx_fsr1.h)
file(WRITE ${CMAKE_BINARY_DIR}/generated/ffx_fsr1_embed.h
     "// Generated by src/CMakeLists.txt from third_party/ffx_fsr1 (AMD FidelityFX FSR 1, MIT): do not edit.\n#pragma once\n"
     "static const unsigned char kFfxA[] = {\n${FFX_A_BYTES}0};\nstatic const unsigned char kFfxFsr1[] = {\n${FFX_FSR1_BYTES}0};\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FFX_FSR1_DIR}/ffx_a.h ${FFX_FSR1_DIR}/ffx_fsr1.h)
target_include_directories(platform PRIVATE ${CMAKE_BINARY_DIR}/generated)

# NVIDIA DLSS Super Resolution: mua3_dlss.dll (src/dlss), which the game loads at run time (gpu/d3d11/dlss.cpp), so
# the game and its libraries never link NVIDIA's code. NVIDIA's SDK is proprietary and never committed: the DLL is
# built only when third_party/dlss (git-ignored) holds its headers (include/) and static-CRT import library
# (lib/nvsdk_ngx_s.lib, from github.com/NVIDIA/DLSS); lib/nvngx_dlss.dll, if there, is copied next to it.
#   MUA3_DLSS=OFF leaves it out even then (release packages).
option(MUA3_DLSS "Build mua3_dlss.dll when NVIDIA's SDK is in third_party/dlss" ON)
set(DLSS_DIR ${CMAKE_SOURCE_DIR}/third_party/dlss)
if(MUA3_DLSS AND EXISTS ${DLSS_DIR}/include/nvsdk_ngx.h AND EXISTS ${DLSS_DIR}/lib/nvsdk_ngx_s.lib)
    add_library(mua3_dlss SHARED dlss/mua3_dlss.cpp dlss/mua3_dlss_api.h)
    target_include_directories(mua3_dlss PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${DLSS_DIR}/include)
    target_link_libraries(mua3_dlss PRIVATE ${DLSS_DIR}/lib/nvsdk_ngx_s.lib d3d11)
    set_target_properties(mua3_dlss PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    if(EXISTS ${DLSS_DIR}/lib/nvngx_dlss.dll)
        add_custom_command(TARGET mua3_dlss POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                           ${DLSS_DIR}/lib/nvngx_dlss.dll $<TARGET_FILE_DIR:mua3_dlss>)
    endif()
    message(STATUS "DLSS: mua3_dlss.dll is built (NVIDIA SDK in third_party/dlss)")
else()
    message(STATUS "DLSS: mua3_dlss.dll not built (no NVIDIA SDK in third_party/dlss)")
endif()
